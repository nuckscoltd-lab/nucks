        // ---------- 알림: 게시판 새 글 + 다가오는 연차(일주일 이내) ----------
        async function checkNotifications(user, role) {
            const notifItems = [];

            try {
                // 1. 게시판 새 글 여부 확인 (본인이 마지막으로 게시판을 확인한 시각 이후 글이 있는지)
                const userDoc = await db.collection('users').doc(user.uid).get();
                const lastSeenBoard = userDoc.data()?.lastSeenBoard;

                const latestPostSnap = await db.collection('posts').orderBy('createdAt', 'desc').limit(1).get();
                if (!latestPostSnap.empty) {
                    const latestPost = latestPostSnap.docs[0].data();
                    const isNew = !lastSeenBoard || (latestPost.createdAt && latestPost.createdAt.toMillis() > lastSeenBoard.toMillis());
                    if (isNew) {
                        document.getElementById('boardBadge').style.display = 'block';
                        notifItems.push(`📋 게시판에 새 글이 있습니다: "${latestPost.title}"`);
                    }
                }

                // 2. 관리자: 앞으로 7일 이내 예정된 연차 확인
                if (role === 'admin') {
                    const today = new Date();
                    const todayStr = today.toISOString().slice(0, 10);
                    const weekLater = new Date(today.getTime() + 7 * 24 * 60 * 60 * 1000);
                    const weekLaterStr = weekLater.toISOString().slice(0, 10);

                    const leavesSnap = await db.collection('leaves').get();
                    const upcoming = leavesSnap.docs
                        .map(d => d.data())
                        .filter(l => l['날짜'] >= todayStr && l['날짜'] <= weekLaterStr);

                    if (upcoming.length > 0) {
                        upcoming.sort((a, b) => (a['날짜'] || '').localeCompare(b['날짜'] || ''));
                        notifItems.push('🏖 일주일 내 예정된 연차:');
                        upcoming.forEach(l => {
                            notifItems.push(`　· ${l['사원명'] || ''} - ${l['날짜']} (${l['연차구분'] || '연차'})`);
                        });
                    }
                }

                if (notifItems.length > 0) {
                    showNotifPopup(notifItems);
                }
            } catch (err) {
                console.error('알림 확인 실패:', err);
            }
        }

        function showNotifPopup(items) {
            // 세션당 한 번만 팝업 (새로고침마다 계속 뜨지 않도록)
            if (sessionStorage.getItem('notifShown')) return;
            sessionStorage.setItem('notifShown', '1');

            const popup = document.getElementById('notifPopup');
            const body = document.getElementById('notifBody');
            body.innerHTML = items.map(i => `<div style="margin-bottom:6px;">${i}</div>`).join('');
            popup.style.display = 'block';
        }
