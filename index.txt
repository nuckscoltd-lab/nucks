        // 초기 설정

        // 메뉴 토글
        function toggleMenu() {
            var navbar = document.getElementById("navbar");
            navbar.classList.toggle("active");
        }

        // 필터링 기능
        const itemsPerPage = 50;
        let currentPage = 1;
        let currentCategory = 'all';

        function filterSelection(category, btnElement) {
            currentCategory = category;
            currentPage = 1;
            var btns = document.getElementsByClassName("filter-btn");
            for (var i = 0; i < btns.length; i++) { btns[i].classList.remove("active"); }
            if(btnElement) btnElement.classList.add("active");
            renderPage();
        }

        function renderPage() {
            var items = document.getElementsByClassName("project-item");
            var filteredItems = [];
            for (var i = 0; i < items.length; i++) {
                items[i].classList.add("hide");
                if (currentCategory === "all" || items[i].getAttribute("data-category") === currentCategory) { 
                    filteredItems.push(items[i]); 
                }
            }
            var startIndex = (currentPage - 1) * itemsPerPage;
            for (var j = startIndex; j < startIndex + itemsPerPage && j < filteredItems.length; j++) {
                filteredItems[j].classList.remove("hide");
            }
            createPagination(filteredItems.length);
        }

        function createPagination(totalItems) {
            var pagination = document.getElementById("pagination");
            pagination.innerHTML = "";
            var totalPages = Math.ceil(totalItems / itemsPerPage);
            if (totalPages <= 1) return;
            for (var i = 1; i <= totalPages; i++) {
                var btn = document.createElement("div");
                btn.innerText = i; btn.classList.add("page-num");
                if (i === currentPage) btn.classList.add("active");
                btn.onclick = function() { currentPage = parseInt(this.innerText); renderPage(); };
                pagination.appendChild(btn);
            }
        }

        function openModal(element) {
            let imagesData = element.getAttribute('data-images');
            currentImages = imagesData ? JSON.parse(imagesData) : [element.querySelector('img').src];
            currentImageIndex = 0;
            showSlide(0);
            document.getElementById("imageModal").style.display = "flex";
        }

        function closeModal() { document.getElementById("imageModal").style.display = "none"; }
        function closeModalOnClickOutside(e) { if (e.target.id === "imageModal") closeModal(); }
        
        function changeSlide(n) {
            currentImageIndex = (currentImageIndex + n + currentImages.length) % currentImages.length;
            showSlide(currentImageIndex);
        }

        function showSlide(index) {
            document.getElementById("modalImg").src = currentImages[index];
            document.getElementById("imageCounter").innerText = (index + 1) + " / " + currentImages.length;
        }

        /* [핵심 기능] 비디오 자연스러운 교차 페이드 재생 */
        const videoList = ["nucks_mood.mp4", "nucks.mp4", "nucks1-1.mp4", "nucks1-2.mp4"];
        let currentIdx = 0;
        const v1 = document.getElementById('video1');
        const v2 = document.getElementById('video2');
        let activeVideo = v1;
        let idleVideo = v2;

        function initVideo() {
            v1.src = videoList[0];
            v1.load();
            v1.play();
            // 다음 영상 미리 준비
            prepareNextVideo();
        }

        function prepareNextVideo() {
            const nextIdx = (currentIdx + 1) % videoList.length;
            idleVideo.src = videoList[nextIdx];
            idleVideo.load();
        }

        // 영상이 끝나기 약 1.5초 전에 교차 페이드 시작
        function checkVideoTime() {
            const fadeTime = 1.5; 
            if (activeVideo.duration - activeVideo.currentTime < fadeTime) {
                // 한 번만 실행되도록 이벤트 제거
                activeVideo.ontimeupdate = null;
                
                // 다음 영상 재생 시작 및 페이드 인
                idleVideo.style.opacity = "1";
                idleVideo.style.zIndex = "2";
                idleVideo.play();
                
                // 현재 영상 페이드 아웃
                activeVideo.style.opacity = "0";
                activeVideo.style.zIndex = "1";

                // 역할 교체
                [activeVideo, idleVideo] = [idleVideo, activeVideo];
                currentIdx = (currentIdx + 1) % videoList.length;

                // 다음 교차 지점 감시 시작
                activeVideo.ontimeupdate = checkVideoTime;
                
                // 다음 영상 대기열 로드
                setTimeout(prepareNextVideo, 2000); 
            }
        }

        v1.ontimeupdate = checkVideoTime;

        window.onload = function() {
            initVideo();
            filterSelection('all', document.querySelector('.filter-btn.active'));
        };
